function drawSymbol(type, text) { ctx.strokeStyle = '#000'; ctx.lineWidth = 2; ctx.fillStyle = '#000'; // Línea de entrada superior ctx.beginPath(); ctx.moveTo(0, 0); ctx.lineTo(0, 15); ctx.stroke(); if (type === 'ACOMETIDA') { ctx.strokeRect(-15, 15, 30, 30); ctx.font = 'bold 9px "Courier New"'; ctx.fillText('kWh', -11, 33); ctx.beginPath(); ctx.moveTo(0, 45); ctx.lineTo(0, 60); ctx.stroke(); } else if (type === 'IGA' || type === 'PIA') { // Contacto abierto sin barritas térmicas ctx.beginPath(); ctx.moveTo(0, 15); ctx.lineTo(8, 27); ctx.moveTo(0, 42); ctx.lineTo(0, 60); ctx.stroke(); ctx.beginPath(); ctx.arc(0, 15, 2, 0, Math.PI*2); ctx.fill(); ctx.beginPath(); ctx.arc(0, 42, 2, 0, Math.PI*2); ctx.fill(); ctx.beginPath(); ctx.moveTo(0, 42); ctx.lineTo(0, 60); ctx.stroke(); } else if (type === 'DIFERENCIAL') { ctx.beginPath(); ctx.moveTo(0, 15); ctx.lineTo(8, 27); ctx.moveTo(0, 42); ctx.lineTo(0, 60); ctx.stroke(); ctx.beginPath(); ctx.arc(0, 15, 2, 0, Math.PI*2); ctx.fill(); ctx.beginPath(); ctx.arc(0, 42, 2, 0, Math.PI*2); ctx.fill(); // Cuadrado de test y realimentación del diferencial ctx.strokeRect(8, 20, 8, 8); ctx.beginPath(); ctx.moveTo(8, 24); ctx.lineTo(0, 36); ctx.lineTo(0, 42); ctx.stroke(); ctx.beginPath(); ctx.arc(0, 36, 2, 0, Math.PI*2); ctx.fill(); ctx.beginPath(); ctx.moveTo(0, 42); ctx.lineTo(0, 60); ctx.stroke(); } else if (type === 'TIERRA') { ctx.beginPath(); ctx.moveTo(0, 0); ctx.lineTo(0, 15); ctx.moveTo(-15, 15); ctx.lineTo(15, 15); ctx.moveTo(-10, 20); ctx.lineTo(10, 20); ctx.moveTo(-5, 25); ctx.lineTo(5, 25); ctx.stroke(); } // Texto en bloque vertical (multilínea con separador |) if (text && type !== 'TIERRA') { ctx.font = 'bold 9px "Courier New"'; let lines = text.split('|'); if (lines.length === 1) { lines = text.split(' '); } let maxWidth = 0; lines.forEach(l => { let m = ctx.measureText(l); if (m.width > maxWidth) maxWidth = m.width; }); ctx.fillStyle = '#ffffff'; ctx.fillRect(18, 12, maxWidth + 6, (lines.length * 11) + 4); ctx.fillStyle = '#000000'; lines.forEach((line, index) => { ctx.fillText(line, 21, 23 + (index * 11)); }); } }
